##In R, try to use methylKit for differential methylation analysis
/groups2/houston_grp/diego/software/R-3.4.0/bin/R
library(methylKit)

#Name of the files
file_names <- read.table("cov-file-list.txt", header = FALSE)

#Name of the samples
sample_names <- read.table("sample-names.txt", header = FALSE)

##Phenotype
liver_vs_headkidney <- c(rep(0,33), rep(1,33))

#Load data
raw_data <- methRead(location = as.list(as.character(file_names$V1)), sample.id = as.list(as.character(sample_names$V1)), assembly = "Ssa", pipeline = "bismarkCoverage", treatment = liver_vs_headkidney, mincov = 10)

##Discards bases that have coverage below 10X and also discards the bases that have more than 99.9th percentile of coverage in each sample
filtered.myobj=filterByCoverage(raw_data,lo.count=10,lo.perc=NULL, hi.count=NULL,hi.perc=99.9)

#Methylation information for regions/bases that are covered in all samples
meth=unite(filtered.myobj, destrand=TRUE)

##Using a sliding window approach
tiles=tileMethylCounts(filtered.myobj,win.size=1000,step.size=1000)
